var results = new List<object>();
int failures = 0;
void Test(string name, Action test)
{
    try { test(); results.Add(new { name, passed = true }); }
    catch (Exception ex) { failures++; results.Add(new { name, passed = false, error = ex.ToString() }); }
}
async Task TestAsync(string name, Func<Task> test)
{
    try { await test().WaitAsync(TimeSpan.FromSeconds(5)); results.Add(new { name, passed = true }); }
    catch (Exception ex) { failures++; results.Add(new { name, passed = false, error = ex.ToString() }); }
}

Test("手动模式下更改低功耗不启用跟随，主动OCR仍可按限速执行", () =>
{
    var h = new FollowHarness(); h.Safety.SetMode(FollowMode.Manual); h.Schedule.Reset(); h.Engine.PauseForBrowse();
    long epoch = h.Safety.Epoch, sequence = 0;
    foreach (var sample in new[] { (Time: 1000L, LowPower: false), (Time: 2000L, LowPower: true), (Time: 2500L, LowPower: false) })
    {
        Require(h.Schedule.TryBegin(sample.Time, epoch, "new", false, true, sample.LowPower, out var ticket) && ticket.Manual, "主动OCR被低功耗设置禁用");
        var observation = new FollowObservation(epoch, ++sequence, "new", true, [new() { Text = "第二句完整台词", Score = .99 }], "section");
        var decision = h.Safety.Evaluate(h.Engine, observation);
        Require(decision.Kind == FollowDecisionKind.Ignored && !h.Safety.TryApply(h.Engine, observation, decision), "手动识别被当成持续跟随");
        Require(h.Schedule.Complete(ticket, true, false, 100), "手动识别结果不能完成");
        Require(h.Safety.Mode == FollowMode.Manual && !h.Safety.IsArmed && h.Safety.Epoch == epoch, "低功耗设置改变手动选择");
        Require(h.Plays == 0 && h.Engine.CurrentId == "first", "手动OCR意外推进或发声");
    }
});
Test("用户手动切模式使在途结果失效，重新跟随需确认位置和全新双次识别", () =>
{
    var h = new FollowHarness();
    Require(h.Read(1000, "new", "第二句完整台词") && h.Safety.HasPendingConfirmation, "未建立第一次确认");
    long oldEpoch = h.Safety.Epoch;
    Require(h.Schedule.TryBegin(1500, oldEpoch, "new", true, false, false, out var oldTicket), "第二次识别未开始");
    var oldObservation = new FollowObservation(oldEpoch, 2, "new", true, [new() { Text = "第二句完整台词", Score = .99 }], "section");
    h.Safety.SetMode(FollowMode.Manual); h.Schedule.Reset(); h.Engine.PauseForBrowse();
    var oldDecision = h.Safety.Evaluate(h.Engine, oldObservation);
    Require(oldDecision.Kind == FollowDecisionKind.Ignored && !h.Safety.TryApply(h.Engine, oldObservation, oldDecision), "切到手动后旧结果仍推进");
    Require(!h.Schedule.Complete(oldTicket, true, true, 700), "旧结果污染新模式的调度状态");
    Require(h.Plays == 0 && h.Engine.CurrentId == "first" && h.Safety.Mode == FollowMode.Manual && !h.Safety.IsArmed, "手动切换未静音保留位置");
    h.Safety.SetMode(FollowMode.Automatic); h.Schedule.Reset();
    Require(!h.Safety.IsArmed && h.Plays == 0, "选择自动跟随后立即恢复播放");
    Require(!h.Safety.TryApply(h.Engine, oldObservation, oldDecision), "重新选自动模式后旧结果复活");
    Require(h.Engine.ConfirmCurrentPosition() && h.Safety.ConfirmPosition(h.Engine, "old"), "明确确认位置后无法开始跟随");
    Require(h.Read(2000, "new", "第二句完整台词") && h.Plays == 0, "重开后的第一次识别沿用旧确认");
    Require(h.Read(2500, "new", "第二句完整台词") && h.Plays == 1 && h.Engine.CurrentId == "second", "重开后全新双次识别不能继续");
});

await TestAsync("空闲时停止也立即返回，耗时模型释放在后台且不阻塞调用线程", async () =>
{
    var lifetime = new OcrWorkLifetime(); var entered = Signal(); var finish = Signal(); var callerFinished = Signal();
    var returned = new TaskCompletionSource<Task>(TaskCreationOptions.RunContinuationsAsynchronously);
    int callingThread = 0, releaseThread = 0;
    var caller = new Thread(() =>
    {
        try
        {
            callingThread = Environment.CurrentManagedThreadId;
            var stop = lifetime.StopAsync(() =>
            {
                releaseThread = Environment.CurrentManagedThreadId; entered.SetResult();
                finish.Task.GetAwaiter().GetResult();
            });
            returned.SetResult(stop);
        }
        catch(Exception ex) { returned.TrySetException(ex); }
        finally { callerFinished.SetResult(); }
    }) { IsBackground = true, Name = "ocr-stop-test-caller" };
    caller.Start();
    try
    {
        await entered.Task;
        var stop = await returned.Task.WaitAsync(TimeSpan.FromSeconds(2));
        Require(!stop.IsCompleted, "阻塞释放未保持在途");
        Require(callingThread != releaseThread, "调用停止的线程直接释放模型");
    }
    finally { finish.TrySetResult(); }
    await callerFinished.Task; await await returned.Task;
});
await TestAsync("停止立即失效已排队的旧请求，无需等待后台释放完成", async () =>
{
    var lifetime = new OcrWorkLifetime(); var running = Signal(); var finish = Signal();
    int initialized = 0, recognized = 0;
    var old = lifetime.RunAsync(lifetime.CreateRequest(), () => Task.CompletedTask,
        async () => { running.SetResult(); await finish.Task; return 1; });
    await running.Task;
    var queued = lifetime.RunAsync(lifetime.CreateRequest(), () => { initialized++; return Task.CompletedTask; },
        () => { recognized++; return Task.FromResult(2); });
    var stop = lifetime.StopAsync(() => { });
    Require(!queued.IsCompleted, "旧请求没有排在在途任务之后");
    finish.SetResult(); await old; await Cancelled(queued); await stop;
    Require(initialized == 0 && recognized == 0, "旧排队请求在停止之后重新初始化");
});

static void Require(bool condition, string message) { if (!condition) throw new InvalidOperationException(message); }
static OcrScheduleTicket Begin(OcrInferenceSchedule s, long now, string frame = "a")
{
    Require(s.TryBegin(now, 1, frame, true, false, false, out var ticket), "应开始推理却被阻止：" + now);
    return ticket;
}
static void Complete(OcrInferenceSchedule s, OcrScheduleTicket ticket) => Require(s.Complete(ticket, true, false, 100), "当前结果未接纳");
static TaskCompletionSource Signal() => new(TaskCreationOptions.RunContinuationsAsynchronously);
static async Task Cancelled(Task task)
{
    try { await task; } catch (OperationCanceledException) { return; }
    throw new InvalidOperationException("旧任务未取消");
}

sealed class FollowHarness
{
    public PlaybackEngine Engine { get; }
    public FollowSafetyController Safety { get; } = new();
    public OcrInferenceSchedule Schedule { get; } = new();
    public int Plays { get; private set; }
    long sequence;
    public FollowHarness()
    {
        var pack = new Pack { Id = "调度测试", SchemaVersion = 3,
            Chapters = [new() { Id = "chapter", Sections = [new() { Id = "section", StartId = "first" }] }],
            Nodes = [new() { Id = "first", SectionId = "section", Text = "第一句完整台词", NextId = "second" },
                new() { Id = "second", SectionId = "section", Text = "第二句完整台词", NextId = "third" },
                new() { Id = "third", SectionId = "section", Text = "第三句完整台词" }] };
        pack.Validate(); Engine = new(pack); Engine.Commit("first"); Engine.PlayRequested += _ => Plays++;
        Safety.SetMode(FollowMode.Automatic); if (!Safety.ConfirmPosition(Engine, "old")) throw new InvalidOperationException("fixture");
    }
    public bool Read(long now, string frame, string text, bool lowPower = false)
    {
        if (!Schedule.TryBegin(now, Safety.Epoch, frame, true, false, lowPower, out var ticket)) return false;
        var observation = new FollowObservation(Safety.Epoch, ++sequence, frame, true, [new() { Text = text, Score = .99 }], "section");
        var decision = Safety.Evaluate(Engine, observation); Safety.TryApply(Engine, observation, decision);
        if (!Schedule.Complete(ticket, true, Safety.HasPendingConfirmation, 100)) throw new InvalidOperationException("有效结果失效");
        return true;
    }
}
