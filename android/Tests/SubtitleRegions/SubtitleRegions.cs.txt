using System.Text.Json;
using PgrVoice.AndroidApp;

var tests = new List<object>(); int failures = 0;
void Test(string name, Action action)
{
    try { action(); tests.Add(new { name, passed = true }); }
    catch (Exception ex) { failures++; tests.Add(new { name, passed = false, error = ex.Message }); }
}
int w = SubtitleRegionSelector.SampleWidth, h = SubtitleRegionSelector.SampleHeight;
int[] Blank(int color = unchecked((int)0xff000000)) => Enumerable.Repeat(color, w * h).ToArray();
bool Detect(int[] pixels) => SubtitleRegionSelector.TrySelectNarration(pixels, w, h, out _);
void Require(bool value, string message) { if (!value) throw new Exception(message); }
void Fill(int[] p, int x, int y, int width, int height, int color = unchecked((int)0xffffffff))
{
    for (int yy = y; yy < y + height; yy++) for (int xx = x; xx < x + width; xx++) p[yy * w + xx] = color;
}
// 小型方块字形，有内部空隙；提供可变行数、位置和长短，不模拟 OCR 语义。
void TextLine(int[] p, int x, int y, int chars = 9, int color = unchecked((int)0xffffffff))
{
    for (int c = 0; c < chars; c++)
    {
        int start = x + c * 6;
        Fill(p, start, y, 4, 1, color); Fill(p, start, y + 4, 4, 1, color);
        Fill(p, start, y, 1, 5, color); Fill(p, start + 3, y, 1, 5, color);
    }
}

foreach (var size in new[] { (320, 192), (160, 96) })
    Test($"真实第29章黑底旁白 {size.Item1}x{size.Item2} 触发中央范围", () =>

        byte[] bytes = File.ReadAllBytes(Path.Combine(AppContext.BaseDirectory, "fixtures", $"chapter29-narration-{size.Item1}x{size.Item2}.argb"));
        int[] pixels = new int[bytes.Length / 4]; Buffer.BlockCopy(bytes, 0, pixels, 0, bytes.Length);
        Require(SubtitleRegionSelector.TrySelectNarration(pixels, size.Item1, size.Item2, out var region), "真实旁白漏判");
        Require(region == new NarrationRegion(.04, .15, .92, .50), "输出范围改变");
        Require(region.Y > .05 && region.Y + region.Height < .69, "选区包含顶部菜单或底部普通对白");
    });

Test("中央白字并伴底部对白时不自动改框", () => { var p = Blank(); TextLine(p, 120, 58); TextLine(p, 45, 160, 35); Require(!Detect(p), "混合布局误判"); });
Test("正常彩色场景对白保持原框", () => { var p = Blank(unchecked((int)0xff305070)); TextLine(p, 50, 160, 30); Require(!Detect(p), "彩色背景误判"); });

Test("暗灰背景中央文字不视为近黑旁白", () => { var p = Blank(unchecked((int)0xff202020)); TextLine(p, 120, 58); Require(!Detect(p), "暗灰场景误判"); });
Test("近黑中央单行文字可识别", () => { var p = Blank(unchecked((int)0xff080808)); TextLine(p, 120, 58); Require(Detect(p), "单行漏判"); });
