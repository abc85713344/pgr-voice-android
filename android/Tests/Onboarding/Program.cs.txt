using System.Text.Json;

var results=new List<object>();int failures=0;
void Check(bool ok,string why){if(!ok)throw new Exception(why);}
void Test(string name,Action run)
{
    try{run();results.Add(new{name,passed=true});Console.WriteLine("PASS "+name);}
    catch(Exception ex){failures++;results.Add(new{name,passed=false,error=ex.ToString()});Console.WriteLine("FAIL "+name+": "+ex.Message);}
}
string artifacts=Path.Combine(Path.GetTempPath(),"pgr-onboarding-"+Guid.NewGuid().ToString("N"));Directory.CreateDirectory(artifacts);
string Area(){string dir=Path.Combine(artifacts,Guid.NewGuid().ToString("N"));Directory.CreateDirectory(dir);return dir;}
OnboardingProgress Trial()
{var p=new OnboardingProgress();p.Begin();p.SelectPurpose(OnboardingPurpose.Listening);p.SelectLine("p","c","n");return p;}
Pack PackFixture(string root)
{
    File.WriteAllText(Path.Combine(root,"sample.wav"),"test");
    var pack=new Pack{Id="p",Title="引导测试",Root=root,
        Chapters=new(){new(){Id="c",Title="章节",Sections=new(){new(){Id="s",Title="小节",StartId="a"}}}},
        Nodes=new(){new(){Id="a",SectionId="s",Kind="line",Speaker="角色甲",Text="第一句",Audio="sample.wav",NextId="b"},
            new(){Id="b",SectionId="s",Kind="line",Speaker="角色乙",Text="第二句",Audio="sample.wav"}}};
    pack.Validate();return pack;
}
void Ready(AppSession s,OnboardingPurpose purpose)
{s.Onboarding.Begin();s.Onboarding.Progress.SelectPurpose(purpose);s.Onboarding.Progress.SelectLine("p","c","a");s.Onboarding.Progress.ConfirmHeard(true);}

Test("新安装判定在设置自动写入前冻结，首次进入后不重复弹出",()=>
{

    Json.Save(Path.Combine(root,"settings.json"),new{volume=.85});
    var store=new OnboardingStore(root,frozen,false);Check(store.OfferAutomatically,"自动保存误判老用户");
    store.Begin();Check(!store.OfferAutomatically,"本进程二次自动弹出");
    Check(!new OnboardingStore(root,OnboardingInstallSnapshot.Capture(root),false).OfferAutomatically,"重启重复弹出");
});

Test("跳过、关闭和完成均持久记录，设置重开可继续或重新开始",()=>
{
    foreach(bool skip in new[]{false,true})
    {string root=Area();var store=new OnboardingStore(root,new(false,false),false);store.Begin();store.Progress.SelectPurpose(OnboardingPurpose.Game);store.Exit(skip);
        store=new(root,new(false,false),false);Check(!store.OfferAutomatically,"退出后再自动弹");store.Begin();Check(store.Progress.Step==OnboardingStep.Package,"重开未保持步骤");}
    var p=Trial();Check(p.ConfirmHeard(true)&&p.Complete(),"合法完成失败");p.Begin();Check(p.Step==OnboardingStep.Purpose&&!p.HeardConfirmed,"完成后重开没有从用途开始");
});
Test("试听前、失败或回退后不能跳过确认完成",()=>
{
    var p=Trial();Check(!p.ConfirmHeard(false)&&!p.Complete(),"未试听可以完成");Check(p.ConfirmHeard(true),"有效试听不能确认");
    p.Back();Check(p.Step==OnboardingStep.Trial&&!p.HeardConfirmed&&!p.Complete(),"回退留下完成权限");
});
Test("文件选择与权限往返的步骤可持久恢复，退出和新一轮屏蔽旧回调",()=>
{
    string root=Area();var store=new OnboardingStore(root,new(false,false),false);store.Begin();store.Progress.SelectPurpose(OnboardingPurpose.Game);
    long epoch=store.Progress.BeginExternal("import");int request=store.Progress.ExternalRequestCode;store.Save();store=new(root,new(false,false),false);
    Check(store.Progress.AcceptExternal("import",epoch)&&store.Progress.Step==OnboardingStep.Package,"返回丢失原步骤");
    Check(!store.Progress.AcceptExternal("overlay",epoch),"错误请求种类被接受");store.Exit(false);
    Check(!store.Progress.AcceptExternal("import",epoch),"关闭后接收旧回调");store.Begin();store.Progress.BeginExternal("import");
    Check(!store.Progress.AcceptExternal("import",epoch),"旧轮次作用于新引导");
    Check(store.Progress.ExternalRequestCode!=request,"不同请求重复使用相同请求码，旧结果可能误触发");
});
Test("单句试听自然结束才可确认，结束不会追加第二次播放",()=>

Test("缺音频、总量/角色静音与系统媒体静音阻止假成功",()=>
{
    bool exists=false,audible=true;float volume=1;var output=new TestAudio();
    using var preview=new OnboardingPreview(output,()=>volume,()=>audible,_=>exists);
    preview.Start("missing");Check(output.Requests==0&&!preview.CanAcknowledge,"缺文件调用播放器");
    exists=true;volume=0;preview.Start("sample");Check(output.Requests==0,"静音仍试听");
    volume=1;audible=false;preview.Start("sample");Check(output.Requests==0,"系统静音仍试听");
    audible=true;preview.Start("sample");output.Finish(output.Ticket);Check(preview.CanAcknowledge,"重试不可恢复");
    volume=0;Check(!preview.CanAcknowledge,"播放后设为静音仍可确认");volume=1;exists=false;Check(!preview.CanAcknowledge,"播放后文件消失仍可确认");
});
Test("同步播放失败、解码错误、音频中断与销毁均不能采纳",()=>
{
    var output=new TestAudio{SyncFailure="同步错误"};var preview=new OnboardingPreview(output,()=>1,()=>true,_=>true);
    preview.Start("a");output.Finish(output.Ticket);Check(!preview.CanAcknowledge,"同步错误被计完成");
    output.SyncFailure=null;preview.Start("a");output.Fail();output.Finish(output.Ticket);Check(!preview.CanAcknowledge,"解码失败被计完成");
    preview.Start("a");output.Interrupt();output.Finish(output.Ticket);Check(!preview.CanAcknowledge,"音频中断被计完成");
    preview.Start("a");preview.Dispose();output.Finish(output.Ticket);Check(!preview.CanAcknowledge,"销毁后回调仍生效");
});

Test("生产入口接线：冻结早于设置写入，引导导入不打开游戏，生命周期暂停试听",()=>
{
    string? app=args.Length>1?args[1]:null;
    for(var dir=new DirectoryInfo(Environment.CurrentDirectory);app==null&&dir!=null;dir=dir.Parent)
        if(File.Exists(Path.Combine(dir.FullName,"android","AppSession.cs")))app=dir.FullName;
    if(app==null)throw new Exception("请将播放器开发根作为第二参数传入");
    string source=File.ReadAllText(Path.Combine(app,"android","AppSession.cs"));
    Check(source.IndexOf("OnboardingInstallSnapshot.Capture(root)")<source.IndexOf("Json.ReadWithBackup<AppSettings>"),"快照发生在读取设置后");
    Check(source.Contains("if(openAfterImport)LoadPack(installed.PackId);"),"普通导入与引导导入未隔离");
    string main=File.ReadAllText(Path.Combine(app,"android","MainActivity.cs"));
    Check(main.Contains("OnPause(){PauseOnboarding();")&&main.Contains("DestroyOnboarding();")&&main.Contains("RestoreOrStartOnboarding(savedInstanceState);"),"生命周期未接线");
    string ui=File.ReadAllText(Path.Combine(app,"android","MainActivity.Onboarding.cs"));
    Check(ui.Contains("openAfterImport:false")&&ui.Contains("Guide.Epoch!=epoch")&&ui.Contains("Guide.AcceptExternal(kind,epoch)"),"导入身份保护缺失");
    Check(!ui.Contains("RequestCapture(")&&!ui.Contains("ExplainAutoPlaybackAccessibility("),"引导自动申请无关权限");
});
