string beforeHash = Convert.ToHexStringLower(SHA256.HashData(File.ReadAllBytes(packPath)));
var results = new List<object>(); int failures = 0;
void Check(bool condition, string message) { if (!condition) throw new Exception(message); }
void Test(string name, Action action)
{
    try { action(); results.Add(new { name, passed = true }); Console.WriteLine("PASS " + name); }
    catch (Exception ex) { failures++; results.Add(new { name, passed = false, error = ex.ToString() }); Console.WriteLine("FAIL " + name + ": " + ex.Message); }
}
AppSession Session(string menuId = RootMenu, bool enabled = true)
{
    Android.OS.SystemClock.Now = 1_000;
    var engine = new PlaybackEngine(Pack.Load(packPath));
    Check(engine.OpenGameMenu(menuId, Section), engine.NavigationError);
    var s = new AppSession(engine); s.Settings.BranchAutoFollow = enabled; s.SectionId = Section;
    s.OpenBrowseForTest(OverlayBrowsePage.Branches); return s;
}
string Navigation(AppSession s) => JsonSerializer.Serialize(s.Engine!.ExportNavigation(), Json.Options);
OverlayBrowseItem OptionItem(AppSession s, ChoiceOption option) => s.BrowseModel.Items.Single(i => i.Title.StartsWith(option.Label, StringComparison.Ordinal));
void BrowseSelect(AppSession s, ChoiceOption option)
{
    var model = s.BrowseModel; s.ChooseBrowseForTest(model, OptionItem(s, option)); s.AdvanceClock(180);
}
void ConfirmPrompt(AppSession s, int index)
{
    s.Window.LatestBranch.Selected(index); s.AdvanceClock(180);
}
void RequireStoppedList(AppSession s, string menuId, int options)
{
    Check(s.Engine!.CurrentId == menuId && s.Engine.Mode == RunMode.Choice, "没有停在预期菜单：" + s.Engine.CurrentId);
    Check(!s.ClickFollowRunning && !s.AutoPlaybackRunning && !s.BranchFollowRunning && TapFollowOverlay.VisibleCount == 0, "菜单出现后仍有推进触窗或自动播放");
    Check(s.Window.ManualBranchVisible && s.Window.LatestBranch.Labels.Count == options && s.Window.LatestBranch.TapWindowsAtShow == 0, "菜单列表缺项或撤窗晚于列表出现");
}
void FinishLineSegment(AppSession s, int maximum = 100)
{
    int count = 0;
    while (s.Engine!.Current is { Kind: "line" })
    {
        if (++count > maximum) throw new Exception("真实段落未在有限步内结束");
        Check(s.ClickFollowRunning, "正文尚未结束却未恢复点按：" + s.Status);
        int requests = s.GestureRequests;
        s.AdvanceClock(350); s.TapNextForTest(); s.AdvanceClock(80);
        Check(s.GestureRequests == requests + 1, "用户轻点没有恰好转发一次");
        GameAdvanceAccessibilityService.CompleteOldestGesture(); s.AdvanceClock(180);
    }
}
void OpenActorsBySecondary(AppSession s)
{
    var prompt = s.Window.LatestBranch;
    Check(prompt.SecondaryAction != null && !string.IsNullOrWhiteSpace(prompt.SecondaryActionLabel), "话题列表没有人工换NPC入口");
    int plays = s.PlayRequestedNodes.Count, taps = s.GestureRequests;
    prompt.SecondaryAction!(); s.AdvanceClock(180);
    RequireStoppedList(s, RootMenu, 8);
    Check(s.PlayRequestedNodes.Count == plays && s.GestureRequests == taps, "换NPC额外发声或点击游戏");
}
ChoiceOption RootOption(AppSession s, int index) => s.Engine!.Pack.ById[RootMenu].Options[index];
void SelectActorWithOpening(AppSession s, int index, bool fromRootPrompt = false)
{
    var option = RootOption(s, index);
    if (fromRootPrompt) ConfirmPrompt(s, index); else BrowseSelect(s, option);
    if (index is 0 or 1 or 2 or 3)
    {
        Check(!s.ClickFollowRunning && s.Engine!.CurrentId == RootMenu && s.Window.LatestBranch.Labels.Count == 2, "多段NPC应先静音选择当前内容");
        ConfirmPrompt(s, 0);
    }
    Check(s.Engine!.CurrentId == option.TargetId && s.ClickFollowRunning, "确认开场后未到对应NPC首句：" + s.Status);
}
