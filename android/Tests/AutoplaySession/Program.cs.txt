var results = new List<object>(); int failures = 0;
void Test(string name, Action test)
{
    try { test(); results.Add(new { name, passed = true }); Console.WriteLine("PASS " + name); }
    catch (Exception ex) { failures++; results.Add(new { name, passed = false, error = ex.ToString() }); Console.WriteLine("FAIL " + name + ": " + ex.Message); }
}
void Check(bool value, string message) { if (!value) throw new Exception(message); }
AppSession Session(string node = "a", bool currentAudio = true, bool nextAudio = true)
{
    Android.OS.SystemClock.Now = 1_000;
    var engine = new PlaybackEngine(Fixture(currentAudio, nextAudio));
    if (node == "branch") Check(engine.ConfirmGameLine(node), "支线样本必须先人工确认路线");
    else engine.Commit(node);
    engine.PauseForBrowse();
    return new(engine);
}
void Start(AppSession session)
{
    session.BeginFromOverlay(); session.Window.Latest.Confirmed(); session.AdvanceClock(180);
    Check(session.AutoPlaybackRunning, "当前共同句未能启动：" + session.Status);
}
void Silent(AppSession session) => Check(session.PlayedNodes.Count == 0 && session.GestureRequests == 0 && !session.AutoPlaybackRunning, "意外播放、点击或启动");
AppSession BranchSession(Action<Pack>? configure = null)
{
    Android.OS.SystemClock.Now = 1_000;
    var pack = BranchFixture(); configure?.Invoke(pack); pack.Validate();
    var engine = new PlaybackEngine(pack); engine.Commit("common0"); engine.PauseForBrowse();
    var session = new AppSession(engine); session.Settings.BranchAutoFollow = true; return session;
}
void EnterBranchMenu(AppSession s)
{
    Start(s); long audio = s.BoundAudioTicket; s.CompleteAudio(audio); s.CompleteAudio(audio);
    Check(s.GestureRequests == 1 && s.Engine!.CurrentId == "common0", "音频结束未等待一次手势，或重复补点");
    GameAdvanceAccessibilityService.CompleteOldestGesture(); s.AdvanceClock(450);
    Check(!s.AutoPlaybackRunning && !s.BranchFollowRunning && !s.ClickFollowRunning && s.Engine!.CurrentId == "menu" && s.Window.BranchConfirmations.Count == 1, "未由真实自动播放进入手动分支确认：" + s.Status);
    Check(s.Engine.Choices.Count == 0 && s.PlayedNodes.SequenceEqual(["common0"]), "只进菜单就提交路线或播放支线");
}
void EnterBranchLine(AppSession s, string side = "left")
{
    EnterBranchMenu(s);
    s.Window.LatestBranch.Selected(side == "left" ? 0 : 1); s.AdvanceClock(180);
    Check(s.ClickFollowRunning && !s.AutoPlaybackRunning && !s.BranchFollowRunning && s.Engine!.CurrentId == side + "1", "分支确认后未只恢复点按跟随");
    Check(s.PlayedNodes.SequenceEqual(["common0", side + "1"]) && s.GestureRequests == 1, "分支首句重复播、被跳过或额外点按");
}
void TapToCommonPrompt(AppSession s)
{
    int before = s.GestureRequests; s.TapNextForTest(); s.AdvanceClock(80);
    Check(s.GestureRequests == before + 1, "点按跟随未发起一次用户手势");
    GameAdvanceAccessibilityService.CompleteOldestGesture();
    Check(s.Engine!.CurrentId == "merge" && s.Engine.Mode == RunMode.Merge && !s.ClickFollowRunning && !s.AutoPlaybackRunning &&
        s.Window.Latest.IsCommonReturn, "汇合后没有停在真实 merge 并显示共同线专用确认：" + s.Status);
}
