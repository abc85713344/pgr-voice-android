var results = new List<object>(); int failures = 0;
void Test(string name, Action test)
{
    try { test(); results.Add(new { name, passed = true }); Console.WriteLine("PASS " + name); }
    catch (Exception ex) { failures++; results.Add(new { name, passed = false, error = ex.ToString() }); Console.WriteLine("FAIL " + name + ": " + ex.Message); }
}
void Check(bool value, string message) { if (!value) throw new Exception(message); }
AppSession Session(string node = "a", bool currentAudio = true, bool nextAudio = true)
{
    Android.OS.SystemClock.Now = 1_000;
    var engine = new PlaybackEngine(Fixture(currentAudio, nextAudio));
    if (node == "branch") Check(engine.ConfirmGameLine(node), "支线样本必须先人工确认路线");
    else engine.Commit(node);
    engine.PauseForBrowse();
    return new(engine);
}
void Start(AppSession session)
{
    session.BeginFromOverlay(); session.Window.Latest.Confirmed(); session.AdvanceClock(180);
    Check(session.AutoPlaybackRunning, "当前共同句未能启动：" + session.Status);
}
void Silent(AppSession session) => Check(session.PlayedNodes.Count == 0 && session.GestureRequests == 0 && !session.AutoPlaybackRunning, "意外播放、点击或启动");

Test("连续十轮取消、手动OCR、重试均仅接受当轮新确认", () => {
    var s = Session(); int expected = 0;
    for (int i = 0; i < 10; i++)
    {
        s.BeginFromOverlay(); var cancelled = s.Window.Latest;
        cancelled.Confirmed(); cancelled.Cancelled(); s.AdvanceClock(180);
        Check(s.PlayedNodes.Count == expected && !s.AutoPlaybackRunning, "取消轮启动了播放");
        s.RequestOcr(true); s.Engine!.ConfirmCurrentPosition();
        s.BeginFromOverlay(); var current = s.Window.Latest;
        cancelled.Confirmed(); cancelled.Cancelled(); current.Confirmed(); current.Confirmed(); s.AdvanceClock(180);
        expected++; Check(s.PlayedNodes.Count == expected && s.AutoPlaybackRunning, "重试未起播或重复起播");
        long oldAudio = s.BoundAudioTicket; s.StopAutoPlayback(); s.CompleteAudio(oldAudio);
        Check(s.GestureRequests == 0 && !s.AutoPlaybackRunning, "取消后的旧音频补点");
    }
    Check(s.OcrRequests == 10, "自动起点额外启动了OCR");
});

static Pack Fixture(bool currentAudio, bool nextAudio)
{
    string root = Path.GetFullPath(Path.Combine("android/Tests/AutoplaySession/artifacts", Guid.NewGuid().ToString("N")));
    Directory.CreateDirectory(root);
    foreach (string file in new[] { "a.wav", "b.wav", "branch.wav" }) File.WriteAllBytes(Path.Combine(root, file), [0]);
    var p = new Pack { Root = root, Id = "autoplay-session-fixture", SchemaVersion = 1,
        Chapters = [new() { Id = "c", Sections = [new() { Id = "s", Title = "第一节", StartId = "a" }] }],
        Nodes = [
            new() { Id = "a", SectionId = "s", Speaker = "角色", Text = "第一句", NextId = "b", Audio = currentAudio ? "a.wav" : null },
            new() { Id = "b", SectionId = "s", Text = "共同线末句", NextId = "menu", Audio = nextAudio ? "b.wav" : null },
            new() { Id = "menu", SectionId = "s", Kind = "choice", Text = "请选择", Options = [new() { Id = "o", PathId = "p", Label = "进入支线", TargetId = "branch", MergeId = "merge", Verified = true }] },
            new() { Id = "branch", SectionId = "s", PathId = "p", Text = "支线对白", NextId = "merge", Audio = "branch.wav" },
            new() { Id = "merge", SectionId = "s", Kind = "merge" },
            new() { Id = "gap", SectionId = "s", Kind = "gap" }
        ]
    };
    p.Validate(); return p;
}
