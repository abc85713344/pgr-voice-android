var results = new List<object>(); int failures = 0;
void Check(bool condition, string message) { if (!condition) throw new Exception(message); }
void Test(string name, Action action)
{
    try { action(); results.Add(new { name, passed = true }); Console.WriteLine("PASS " + name); }
    catch (Exception ex) { failures++; results.Add(new { name, passed = false, error = ex.ToString() }); Console.WriteLine("FAIL " + name + ": " + ex.Message); }
}
AppSession Session(bool followBranch = false, Action<Pack>? configure = null)
{
    Android.OS.SystemClock.Now = 1_000;
    var pack = OptionalBranchFixtures.Create(); configure?.Invoke(pack); pack.Validate();
    var engine = new PlaybackEngine(pack); engine.Commit("before"); engine.PauseForBrowse();
    var s = new AppSession(engine); s.Settings.BranchAutoFollow = followBranch; return s;
}
string Navigation(AppSession s) => JsonSerializer.Serialize(s.Engine!.ExportNavigation(), Json.Options);
void StartAuto(AppSession s)
{
    s.BeginFromOverlay(); s.Window.Latest.Confirmed(); s.AdvanceClock(180);
    Check(s.AutoPlaybackRunning && s.PlayRequestedNodes.SequenceEqual(["before"]), "前句未真实开启自动播放：" + s.Status);
}
void EnterWait(AppSession s, bool automatic)
{
    if (automatic) { StartAuto(s); s.CompleteAudio(); }
    else
    {
        s.BeginClickFollowForTest(); s.TapNextForTest(); s.AdvanceClock(80);
        Check(s.GestureRequests == 1 && s.Engine!.CurrentId == "before", "点按未等待一次用户手势完成");
        GameAdvanceAccessibilityService.CompleteOldestGesture();
    }
    Check(!s.AutoPlaybackRunning && !s.ClickFollowRunning && !s.BranchFollowRunning && TapFollowOverlay.VisibleCount == 0, "等待期间仍自动/点按跟随");
    Check(s.Engine!.CurrentId == (automatic ? "before" : "menu") && s.GestureRequests == (automatic ? 0 : 1), "暂停模式自动跨进菜单或导航错位");
    Check(s.Window.ManualBranchVisible && s.Window.LatestBranch.Labels.Count == 1 && s.Window.LatestBranch.TapWindowsAtShow == 0, "没有静音显示共同线单项确认或未先撤触窗");
    Check(s.Engine.Choices.Count == 0 && s.Engine.Facts.Count == 0 && s.Engine.Heard.Count == 0, "等待阶段伪造了分支经历");
}
void ConfirmWait(AppSession s)
{
    s.Window.LatestBranch.Selected(0); s.AdvanceClock(180);
}
void CheckNoBranchHistory(AppSession s)
{
    Check(!s.PlayRequestedNodes.Any(id => id.StartsWith("left") || id.StartsWith("right")), "播放了本应暂停的支线");
    Check(s.Engine!.Choices.Count == 0 && s.Engine.Facts.Count == 0 && s.Engine.Heard.Count == 0, "恢复共同线时补写分支选择/任务/已聊状态");
}

Test("缺出口、不同汇合、条件、事实、循环、嵌套和互动菜单不得猜共同线", () => {
    foreach (string scenario in new[] { "unverified", "evidence", "different-merge", "requires", "excludes", "facts", "cycle", "nested", "interaction", "topics", "schema", "archived", "cross-section" })
    {
        var s = Session(); var p = s.Engine!.Pack; var menu = p.ById["menu"];
        s.Engine.OpenGameMenu("menu", "s");
        if (scenario == "unverified") menu.Options[1].ExitVerified = false;
        if (scenario == "evidence") menu.Options[1].ExitEvidence.Clear();
        if (scenario == "different-merge") menu.Options[1].MergeId = "other-merge";
        if (scenario == "requires") menu.Options[1].Requires.Add("task.ready");
        if (scenario == "excludes") menu.Options[1].Excludes.Add("task.done");
        if (scenario == "facts") p.ById["left2"].SetFacts.Add("task.done");
        if (scenario == "cycle") p.ById["right2"].NextId = "right1";
        if (scenario == "nested") p.ById["right2"].Kind = "choice";
        if (scenario is "interaction" or "topics") menu.MenuType = scenario;
        if (scenario == "schema") p.SchemaVersion = 1;
        if (scenario == "archived") p.ById["common1"].Archived = true;
        if (scenario == "cross-section") p.ById["common1"].SectionId = "other";
        string snapshot = Navigation(s);
        Check(!BranchCommonResumePolicy.TryGetCommonLine(s.Engine, "menu", out _, out _), "不可靠图仍给恢复候选：" + scenario);
        Check(Navigation(s) == snapshot && s.PlayRequestedNodes.Count == 0, "拒绝时改变导航：" + scenario);
    }
});
