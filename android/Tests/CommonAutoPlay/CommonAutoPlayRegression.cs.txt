using System.Security.Cryptography;

var results = new List<object>(); var realPacks = new List<object>(); int failures = 0;
void Test(string name, Action test)
{
    try { test(); results.Add(new { name, passed = true }); }
    catch (Exception ex) { failures++; results.Add(new { name, passed = false, error = ex.ToString() }); }
}
for (int schema = 1; schema <= 3; schema++)
{
    int version = schema;
    Test($"schema{version} 普通共同句之间可以申请点击，末句仍可播放但停在选择前", () =>
    {
        var pack = Fixture(version); var e = At(pack, "a");
        var d = e.EvaluateCommonAutoPlayNext(); Require(d.CurrentIsCommon && d.Allowed && d.Next?.Id == "b", "共同线被全部禁用");
        e.Commit("b"); d = e.EvaluateCommonAutoPlayNext(); Require(d.CurrentIsCommon && !d.Allowed && d.Code == "choice", "分支前没有停止或本句不可听");
    });
    Test($"schema{version} 已核实分支入口、内部、出口均不能开启自动播放", () =>
    {
        var pack = Fixture(version);
        foreach (string id in new[] { "branch-a", "branch-b", "branch-c" })
        {
            var e = new PlaybackEngine(pack); Require(e.ConfirmGameLine(id), "分支测试未实际进入");
            var d = e.EvaluateCommonAutoPlayNext(); Require(!d.CurrentIsCommon && !d.Allowed, "分支可自动播放：" + id);
        }
    });
    Test($"schema{version} 手动确认共同汇合之后可重新开始，不跨越merge", () =>
    {
        var pack = Fixture(version); var e = At(pack, "common-after");
        Require(e.EvaluateCommonAutoPlayNext() is { CurrentIsCommon: true, Allowed: true, Next.Id: "common-last" }, "汇合后共同线不可用");
        e.Commit("merge"); Require(!e.EvaluateCommonAutoPlayNext().CurrentIsCommon, "merge被当作普通句");
    });
    Test($"schema{version} 只读评估不触发声音、不推进、不改变条件历史", () =>
    {
        var e = At(Fixture(version), "a"); int plays = 0, stops = 0, changed = 0;
        e.PlayRequested += _ => plays++; e.StopRequested += () => stops++; e.Changed += () => changed++;
        string before = JsonSerializer.Serialize(e.ExportNavigation(), Json.Options);
        for (int i = 0; i < 4; i++) Require(e.EvaluateCommonAutoPlayNext().Allowed, "正常门控失败");
        Require(before == JsonSerializer.Serialize(e.ExportNavigation(), Json.Options) && plays + stops + changed == 0, "门控有副作用");
    });
}

foreach (string chapter in new[] { "第03章_终末展览", "第29章_源解信标" })
{
    string title = chapter;
    Test(title + "真实包逐句共同线和分支门控", () =>
    {
        string path = Path.Combine(packRoot, title, "pack.json"); var pack = Pack.Load(path);
        int common = 0, permitted = 0, branchLines = 0, rejectedBranches = 0, terminalCommon = 0;
        var codes = new Dictionary<string, int>(); var examples = new List<object>();
        foreach (var node in pack.Nodes.Where(n => !n.Archived && n.Kind == "line"))
        {
            var e = new PlaybackEngine(pack); e.ConfirmGameLine(node.Id);
            var d = e.EvaluateCommonAutoPlayNext(); codes[d.Code] = codes.GetValueOrDefault(d.Code) + 1;
            if (d.CurrentIsCommon) common++;
            if (d.CurrentIsCommon && !d.Allowed) terminalCommon++;
            if (d.Allowed)
            {
                permitted++; Require(d.CurrentIsCommon && d.Next != null && d.Next.PathId.Length == 0 && node.NextId == d.Next.Id && d.Next.SectionId == node.SectionId, "非共同直接连接被放行");
                if (examples.Count < 3) examples.Add(new { current = node.Id, next = d.Next!.Id });
            }
            if (node.PathId.Length > 0) { branchLines++; if (!d.CurrentIsCommon && !d.Allowed) rejectedBranches++; }
        }
        Require(permitted > 0 && branchLines > 0 && branchLines == rejectedBranches, "真实包共同线不可用或分支漏放");
        realPacks.Add(new { title = pack.Title, pack.SchemaVersion, source = path, sha256 = Convert.ToHexStringLower(SHA256.HashData(File.ReadAllBytes(path))),
            activeLines = pack.Nodes.Count(n => !n.Archived && n.Kind == "line"), provenCommonLines = common, allowedDirectPairs = permitted,
            commonLinesStoppedAtBoundary = terminalCommon, explicitBranchLines = branchLines, rejectedExplicitBranchLines = rejectedBranches, reasons = codes, examples });
    });
}

static void Require(bool value, string reason) { if (!value) throw new InvalidOperationException(reason); }
static PlaybackEngine At(Pack pack, string id) { var e = new PlaybackEngine(pack); e.Commit(id); return e; }
static Pack Fixture(int schema)
{
    ChoiceOption Option(string id, string path, string target, params string[] lines) => new()
    {
        Id = id, PathId = path, TargetId = target, MergeId = "merge", ReturnId = "merge", Verified = true,
        BodyVerified = true, ExitVerified = true, Evidence = new() { "fixture-proof" }, BodyEvidence = new() { "fixture-body" }, ExitEvidence = new() { "fixture-exit" },
        SegmentIds = lines.ToList(), LineIds = lines.ToList()
    };
    var pack = new Pack { Id = "common-autoplay-" + schema, SchemaVersion = schema,
        Chapters = new() { new() { Id = "c", Sections = new() { new() { Id = "s", StartId = "a" } } } },
        Nodes = new()
        {
            new() { Id = "a", SectionId = "s", Text = "共同开场", NextId = "b" },
            new() { Id = "b", SectionId = "s", Text = "共同末句", NextId = "menu" },
            new() { Id = "menu", SectionId = "s", Kind = "choice", Options = new() { Option("o1", "p1", "branch-a", "branch-a", "branch-b"), Option("o2", "p2", "branch-c", "branch-c") } },
            new() { Id = "branch-a", SectionId = "s", PathId = "p1", Text = "分支开场", NextId = "branch-b" },
            new() { Id = "branch-b", SectionId = "s", PathId = "p1", Text = "分支末句", NextId = "merge" },
            new() { Id = "branch-c", SectionId = "s", PathId = "p2", Text = "第二分支", NextId = "merge" },
            new() { Id = "merge", SectionId = "s", Kind = "merge", NextId = "common-after" },
            new() { Id = "common-after", SectionId = "s", Text = "汇合共同句", NextId = "common-last" },
            new() { Id = "common-last", SectionId = "s", Text = "最后共同句", NextId = "end" },
            new() { Id = "end", SectionId = "s", Kind = "end" }
        } };
    if (schema < 3)
    {
        // 真正按旧版本字段测试，不能依赖新版 SegmentIds 或 Body/ExitVerified 恰好存在。
        foreach (var option in pack.Nodes.SelectMany(n => n.Options))
        {
            option.BodyVerified = false; option.ExitVerified = false; option.SegmentIds.Clear();
            option.BodyEvidence.Clear(); option.ExitEvidence.Clear();
            if (schema == 1) { option.ReturnId = null; option.Evidence.Clear(); option.LineIds.Clear(); }
        }
    }
    pack.Validate(); return pack;
}
